const API_URL = "https://luxurybid-api.kebarmohammed.workers.dev"; const modalBackdrop = document.getElementById("modalBackdrop"); const modalTitle = document.getElementById("modalTitle"); const modalMessage = document.getElementById("modalMessage"); const toast = document.getElementById("toast"); const auctionGrid = document.getElementById("auctionGrid"); const emptyState = document.getElementById("emptyState"); const searchInput = document.getElementById("searchInput"); function openModal(title, message) { modalTitle.textContent = title; modalMessage.textContent = message; modalBackdrop.hidden = false; } function closeModal() { modalBackdrop.hidden = true; } document.getElementById("modalClose").addEventListener("click", closeModal); document.getElementById("modalOk").addEventListener("click", closeModal); modalBackdrop.addEventListener("click", event => { if (event.target === modalBackdrop) closeModal(); }); function showToast(message) { toast.textContent = message; toast.classList.add("show"); setTimeout(() => toast.classList.remove("show"), 1800); } let activeFilter = "all"; let loadedAuctions = []; function getCategory(name) { const text = (name || "").toLowerCase(); if (/phone|computer|laptop|tablet|camera|electronic|tech/.test(text)) { return "tech"; } if (/watch|cloth|shoe|fashion|style|bag|jewel/.test(text)) { return "style"; } return "home"; } function formatETB(value) { const amount = Number(value); if (!Number.isFinite(amount)) return "Price not set"; return "ETB " + amount.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } function makeAuctionCard(auction) { const card = document.createElement("article"); const name = auction.product_name || auction.title || "Auction item"; const category = getCategory(name); card.className = "auction-card"; card.dataset.category = category; card.dataset.name = name.toLowerCase(); const image = document.createElement("div"); image.className = "card-image"; const badge = document.createElement("span"); badge.className = "card-badge"; badge.textContent = "LIVE LISTING"; image.appendChild(badge); if (auction.image_url) { const img = document.createElement("img"); img.src = auction.image_url; img.alt = name; img.loading = "lazy"; img.style.width = "100%"; img.style.height = "100%"; img.style.objectFit = "contain"; img.onerror = () => img.remove(); image.appendChild(img); } else { image.classList.add( category === "tech" ? "phone-bg" : category === "style" ? "watch-bg" : "coffee-bg" ); } const heart = document.createElement("button"); heart.className = "heart"; heart.type = "button"; heart.setAttribute("aria-label", "Save " + name); heart.textContent = "♡"; heart.addEventListener("click", () => { heart.classList.toggle("saved"); heart.textContent = heart.classList.contains("saved") ? "♥" : "♡"; showToast(heart.classList.contains("saved") ? "Saved in this preview" : "Removed from saved preview"); }); image.appendChild(heart); const content = document.createElement("div"); content.className = "card-content"; const categoryLabel = document.createElement("div"); categoryLabel.className = "card-category"; categoryLabel.textContent = category === "tech" ? "TECHNOLOGY" : category === "style" ? "STYLE" : "HOME & LIVING"; const heading = document.createElement("h3"); heading.textContent = name; const description = document.createElement("p"); description.textContent = auction.description || ""; if (description.textContent) { description.style.marginTop = "6px"; } const meta = document.createElement("div"); meta.className = "card-meta"; const metaText = document.createElement("span"); metaText.textContent = "Database listing"; const status = document.createElement("span"); status.className = "status"; status.textContent = "Active"; meta.append(metaText, status); const bottom = document.createElement("div"); bottom.className = "card-bottom"; const priceBlock = document.createElement("div"); const priceLabel = document.createElement("small"); priceLabel.textContent = "Starting price"; const price = document.createElement("strong"); price.textContent = formatETB(auction.starting_price); priceBlock.append(priceLabel, price); const viewButton = document.createElement("button"); viewButton.className = "btn btn-dark bid-btn"; viewButton.type = "button"; viewButton.textContent = "View auction →"; viewButton.addEventListener("click", () => { openModal(name, "This auction is listed in the database, but live bidding and winner calculation are not implemented yet. No bid will be saved and no money will be charged." ); }); bottom.append(priceBlock, viewButton); content.append(categoryLabel, heading); if (description.textContent) content.appendChild(description); content.append(meta, bottom); card.append(image, content); return card; } function filterCards() { const query = searchInput.value.trim().toLowerCase(); let shown = 0; document.querySelectorAll(".auction-card").forEach(card => { const matchesCategory = activeFilter === "all" || card.dataset.category === activeFilter; const matchesSearch = (card.dataset.name || "").includes(query) || card.querySelector("h3").textContent.toLowerCase().includes(query); card.style.display = matchesCategory && matchesSearch ? "" : "none"; if (matchesCategory && matchesSearch) shown++; }); emptyState.hidden = shown !== 0; } async function loadAuctions() { try { const response = await fetch(API_URL + "/api/auctions"); if (!response.ok) throw new Error("API request failed"); const data = await response.json(); if (!data.success || !Array.isArray(data.auctions)) { throw new Error("Unexpected API response"); } loadedAuctions = data.auctions; auctionGrid.replaceChildren(); loadedAuctions.forEach(auction => { auctionGrid.appendChild(makeAuctionCard(auction)); }); filterCards(); if (loadedAuctions.length === 0) { emptyState.textContent = "No active auctions yet. Please check back later."; emptyState.hidden = false; } } catch (error) { console.error("Could not load auctions:", error); emptyState.textContent = "Auctions could not be loaded. Your email has not been saved."); event.target.reset(); }); document.getElementById("menuBtn").addEventListener("click", () => { openModal("Explore LuxuryBid", "Use the sections on this page: Auctions, How it works, and Trust & safety."); }); loadAuctions();
